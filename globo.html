// Confere se o buffer parece um segmento MPEG-TS inteiro
function isValidTs(data) {
const TS = 188;
if (!data || data.byteLength < TS * 50 || data.byteLength % TS !== 0) return false;
const bytes = new Uint8Array(data);
const packets = bytes.length / TS;
return [0, 1, Math.floor(packets / 2), packets - 1].every(i => bytes[i * TS] === 0x47);
}

async function load(reset = true) {
loaded = true;
if (reset) state.retries = 0;
set({
error: false,
loading: true
});

// Nativo roda antes de qualquer await: o play() precisa sair ainda dentro do clique,
// senão o iOS bloqueia (e sem play ele nem carrega o vídeo, fica só "Sintonizando")
if (useNative()) {
console.info('[player] Usando HLS nativo');
return loadNative();
}

const HlsWithP2P = await p2pReady;

if (window.Hls && Hls.isSupported()) {
hls?.destroy();
p2pStats.peers = 0;
const useP2P = HlsWithP2P && !p2pOff;
const HlsClass = useP2P ? HlsWithP2P : Hls;
hls = new HlsClass({
...(useP2P ? {
p2p: p2pConfig()
} : {}),
lowLatencyMode: true,
// Começa N segmentos atrás do ao vivo, conforme o modo de delay escolhido
// (desempenho = 4, equilibrado = 2). Se travar, só espera (loading) e continua
// do mesmo ponto, sem pular para o ao vivo; para isso existe o "Voltar ao vivo".
liveSyncDurationCount: DELAY_MODES[state.delay].segments,
maxBufferLength: 30,
backBufferLength: 60,
manifestLoadingMaxRetry: 1,
levelLoadingMaxRetry: 2,
});

hls.on(Hls.Events.MANIFEST_PARSED, (_, data) => {
state.levels = data.levels
.map((lv, i) => ({
index: i,
height: lv.height,
label: lv.height ? `${lv.height}p` : `${Math.round(lv.bitrate / 1000)} kbps`
}))
.filter(lv => lv.height || lv.label);
renderMenu();
render();
startPlayback();
});

hls.on(Hls.Events.LEVEL_SWITCHED, (_, data) => {
set({
currentHeight: hls.levels[data.level]?.height || null
});
});

hls.on(Hls.Events.ERROR, (_, data) => {
if (data.fatal) console.warn('[player] Erro HLS:', data.type, data.details, data.response?.code ?? '', data.error?.message ?? '');
// Travou por falta de buffer: só mostra o loading, o hls.js retoma sozinho
if (!data.fatal) {
if (data.details === Hls.ErrorDetails.BUFFER_STALLED_ERROR) set({
loading: true
});
return;
}

// Só mostra o loading se o vídeo realmente parou: com buffer, ele segue tocando
if (video.readyState < 3) set({
loading: true
});

if (data.type === Hls.ErrorTypes.MEDIA_ERROR) {
if (state.retries++ >= 3) return fail();
return hls.recoverMediaError();
}

// Erro de rede com o vídeo já tocando: volta a baixar do mesmo ponto, sem
// recriar o player. Continua tentando por ~1 min antes de mostrar o erro.
if (state.started) {
if (state.retries++ >= 12) return fail();
return setTimeout(() => hls?.startLoad(), 5000);
}

// Ainda não tocou (playlist não carregou): recria com espera crescente
if (state.retries++ >= 3) return fail();
setTimeout(() => load(false), 1500 * state.retries);
});



hls.loadSource(src);
hls.attachMedia(video);
} else {
console.warn('[player] Navegador sem suporte a HLS');
fail();
}
}

// Prefere o hls.js (no iOS 17.1+ ele roda via ManagedMediaSource). O nativo fica só para
// quem não tem MSE (iOS antigo): os segmentos vêm como application/javascript e o
// player nativo pode recusar (erro 4), o hls.js não liga para o tipo.
function useNative() {
const native = !!video.canPlayType('application/vnd.apple.mpegurl');
return native && !(window.Hls && Hls.isSupported());
}

// HLS nativo (Safari / iOS)
function loadNative() {
hls?.destroy();
hls = null;
video.removeEventListener('error', onNativeError);
video.addEventListener('error', onNativeError, {
once: true
});
video.src = src;
startPlayback();
}

function onNativeError() {
const err = video.error;
console.warn('[player] Erro no HLS nativo:', err?.code, err?.message || '');
if (state.retries++ >= 3) return fail();
setTimeout(() => load(false), 1500 * state.retries);
}

// Só roda depois do clique em "Assistir", então o navegador permite tocar com som
function startPlayback() {
video.muted = false;
if (state.volume === 0) state.volume = 1;
video.volume = state.volume;
// Se ainda assim o navegador bloquear, tira o loading e mostra o play do meio de novo
video.play().catch(() => set({
loading: false
}));
}

function fail() {
set({
error: true,
loading: false,
playing: false
});
}

// ---------------------------------------------------------------------------
// Anúncio VAST (pré-roll). Toca antes da live, no primeiro clique em "Assistir".
//
// Eventos disparados em window (use addEventListener no seu script):
// 'vast:start' detail: { url, clickUrl, duration }
// 'vast:finished' detail: { status } -> 'complete' | 'skipped' | 'empty' | 'error' | 'timeout'
// Também chama window.onVastFinished(detail), se existir.
// ---------------------------------------------------------------------------
function emitVast(name, detail = {}) {
console.info(`[vast] ${name}`, detail);
try {
window.dispatchEvent(new CustomEvent(name, {
detail
}));
if (name === 'vast:finished' && typeof window.onVastFinished === 'function') window.onVastFinished(detail);
} catch (e) {
console.warn('[vast] Erro no listener:', e);
}

if (name == 'vast:finished') {
open85();
}
}


const vastMacros = url => url
.replace(/\[(CACHEBUSTING|RANDOM)\]|%%(CACHEBUSTING)%%/g, () => Math.floor(Math.random() * 1e9))
.replace(/\[TIMESTAMP\]/g, encodeURIComponent(new Date().toISOString()))
.replace(/\[ERRORCODE\]/g, '900')
.replace(/\[CONTENTPLAYHEAD\]|\[ASSETURI\]/g, '');

const ping = url => {
if (url) new Image().src = vastMacros(url);
};

// "00:00:05" ou "00:00:05.500" -> segundos
function parseTime(t) {
const m = /^(\d+):(\d+):(\d+(?:\.\d+)?)$/.exec((t || '').trim());
return m ? (+m[1]) * 3600 + (+m[2]) * 60 + parseFloat(m[3]) : null;
}

async function fetchVast(url, depth = 0, acc = {
impressions: [],
errors: [],
tracking: {},
clickTracking: []
}) {
if (depth > 5) throw new Error('wrapper demais');
const ctrl = new AbortController();
const timer = setTimeout(() => ctrl.abort(), 8000);
let text;
try {
const res = await fetch(vastMacros(url), {
credentials: 'omit',
signal: ctrl.signal
});
text = await res.text();
} finally {
clearTimeout(timer);
}

const doc = new DOMParser().parseFromString(text, 'text/xml');
const ad = doc.querySelector('Ad');
if (!ad) return null; // VAST vazio

const txt = el => (el?.textContent || '').trim();
ad.querySelectorAll('Impression').forEach(el => acc.impressions.push(txt(el)));
ad.querySelectorAll('Error').forEach(el => acc.errors.push(txt(el)));
ad.querySelectorAll('Tracking').forEach(el => {
(acc.tracking[el.getAttribute('event')] ||= []).push(txt(el));
});
ad.querySelectorAll('ClickTracking').forEach(el => acc.clickTracking.push(txt(el)));

const wrapper = ad.querySelector('Wrapper');
if (wrapper) {
const next = txt(wrapper.querySelector('VASTAdTagURI'));
return next ? fetchVast(next, depth + 1, acc) : null;
}

const linear = ad.querySelector('InLine Linear');
if (!linear) return null;

// Prefere mp4 progressivo, com altura mais próxima de 720p. Ignora VPAID/flash.
const files = [...linear.querySelectorAll('MediaFile')]
.filter(el => !/vpaid/i.test(el.getAttribute('apiFramework') || ''))
.map(el => ({
url: txt(el),
type: el.getAttribute('type') || '',
height: parseInt(el.getAttribute('height')) || 0,
}))
.filter(f => f.url && (/video\/(mp4|webm)/.test(f.type) || (/mpegurl/i.test(f.type) && useNative())));
if (!files.length) return null;
files.sort((a, b) => (/mp4/.test(b.type) - /mp4/.test(a.type)) || (Math.abs(a.height - 720) - Math.abs(b.height - 720)));

return {
...acc,
media: files[0].url,
duration: parseTime(txt(linear.querySelector('Duration'))),
skipOffset: linear.getAttribute('skipoffset'),
clickUrl: txt(linear.querySelector('ClickThrough')),
clickTracking: acc.clickTracking,
};
}

// Toca o anúncio e resolve quando termina (sempre resolve, nunca rejeita)
function runAd() {
return new Promise(async resolve => {
let done = false;
let ad = null;
const fired = new Set();
const track = ev => (ad?.tracking[ev] || []).forEach(ping);
const once = ev => {
if (!fired.has(ev)) {
fired.add(ev);
track(ev);
}
};

const finish = (status, extra = {}) => {
if (done) return;
done = true;
clearTimeout(startTimer);
adVideo.pause();
adVideo.removeAttribute('src');
adVideo.load();
set({
adPlaying: false
});
if (status === 'complete') once('complete');
if (status === 'skipped') track('skip');
if (status === 'error' || status === 'timeout')(ad?.errors || []).forEach(ping);
emitVast('vast:finished', {
status,
...extra
});
resolve(status);
};

const adVideo = ref.adVideo;
let startTimer = null;

try {
ad = await fetchVast(VAST_URL);
} catch (err) {
console.warn('[vast] Falha ao buscar o VAST:', err);
return finish('error', {
error: String(err)
});
}
if (!ad) return finish('empty');

// Se o anúncio não começar a tocar em 8s, segue para a live
startTimer = setTimeout(() => finish('timeout'), 8000);

const skipAt = !ad.skipOffset ? null :
ad.skipOffset.endsWith('%') ?
() => (parseFloat(ad.skipOffset) / 100) * adVideo.duration :
() => parseTime(ad.skipOffset);

adSkipHandler = () => {
const at = skipAt?.();
if (at != null && adVideo.currentTime >= at) finish('skipped');
};
adClickHandler = () => {
if (!ad.clickUrl) return;
ad.clickTracking.forEach(ping);
adVideo.pause();
window.open(vastMacros(ad.clickUrl), '_blank');
};

ref.adLearn.toggleAttribute('hidden', !ad.clickUrl);
ref.adSkip.setAttribute('hidden', '');

adVideo.onerror = () => finish('error', {
error: 'media'
});
adVideo.onended = () => finish('complete');
adVideo.onplaying = () => {
clearTimeout(startTimer);
if (!fired.has('impression')) {
fired.add('impression');
ad.impressions.forEach(ping);
once('start');
once('creativeView');
set({
adPlaying: true,
loading: false
});
emitVast('vast:start', {
url: ad.media,
clickUrl: ad.clickUrl,
duration: adVideo.duration
});
}
};
// Pausado: o clique retoma. Tocando: o clique abre o link do anúncio (e pausa)
adVideo.onclick = () => {
if (adVideo.paused) adVideo.play().catch(() => {});
else adClickHandler();
};
const syncAdIcons = () => {
ref.adIconPlay.toggleAttribute('hidden', !adVideo.paused);
ref.adIconPause.toggleAttribute('hidden', adVideo.paused);
};
adVideo.onplay = syncAdIcons;
adVideo.onpause = () => {
syncAdIcons();
if (!done && !adVideo.ended) track('pause');
};
adVideo.ontimeupdate = () => {
const d = adVideo.duration;
const t = adVideo.currentTime;
if (d > 0) {
if (t >= d * .25) once('firstQuartile');
if (t >= d * .5) once('midpoint');
if (t >= d * .75) once('thirdQuartile');
ref.adTime.textContent = `${Math.max(0, Math.ceil(d - t))}s`;
}
if (skipAt) {
const at = skipAt();
const btn = ref.adSkip;
if (at != null && !isNaN(at)) {
btn.removeAttribute('hidden');
btn.textContent = t >= at ? 'Pular anúncio ›' : `Pular em ${Math.ceil(at - t)}s`;
btn.disabled = t < at;
}
}
};

adVideo.muted = false;
adVideo.src = ad.media;
try {
await adVideo.play();
} catch (e) {
// Autoplay com som bloqueado (ex.: iOS): tenta mudo
adVideo.muted = true;
try {
await adVideo.play();
} catch (e2) {
finish('error', {
error: 'autoplay'
});
}
}
});
}

// É uma live: depois de pausar, o play reinicia o sinal no ponto ao vivo
// em vez de continuar de onde parou
function play() {

// Primeiro clique: só agora começa a baixar o sinal (antes passa pelo anúncio VAST)
if (!loaded) {
// "Destrava" o vídeo ainda dentro do clique: no iOS, o play() com som que vem

return startWithAd();

if (state.started && video.paused) return load();

}

function liveEdge() {
return hls?.liveSyncPosition ?? (video.seekable.length ? video.seekable.end(video.seekable.length - 1) : null);
}

// Considera "ao vivo" se estiver a menos de ~3 segmentos do fim do buffer ao vivo
function checkLive() {
const edge = liveEdge();
if (edge == null) return;
const atLive = edge - video.currentTime < 12;
if (atLive !== state.atLive) set({
atLive
});
}

function goLive() {
const edge = liveEdge();
if (edge != null && edge - video.currentTime > 12) video.currentTime = edge;
}

function setLevel(i) {
set({
level: i,
menu: null
});
if (hls) hls.currentLevel = i === -1 ? -1 : state.levels[i].index;
}

function setDelay(mode) {

// Pausado ou ainda não iniciado, aplica no próximo play (que já recarrega a live).
if (changed && state.playing) load();
}
